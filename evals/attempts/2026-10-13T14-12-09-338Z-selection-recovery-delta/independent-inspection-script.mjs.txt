import {readFile,writeFile,cp} from 'node:fs/promises';
import {createHash} from 'node:crypto';
import {execFileSync} from 'node:child_process';
import {isDeepStrictEqual as eq} from 'node:util';
import {inventory} from '../../evals/inventory.mjs';
import {inspectPlanningFixture} from '../../evals/planning-fixture.mjs';
const attempt='evals/attempts/2026-10-13T14-12-09-338Z-selection-recovery-delta';
const read=async p=>JSON.parse(await readFile(p,'utf8'));
const save=async (p,v)=>writeFile(p,JSON.stringify(v,null,2)+'\n');
const sha=b=>createHash('sha256').update(b).digest('hex');
const ctx=await read(attempt+'/context.json'),freeze=await read(attempt+'/freeze.json');
const checks=[];
for(const {stage,manifest,vaultRoot} of ctx.fixtures){
 const f=await read(manifest),observation=await inspectPlanningFixture(f);
 const before=await read(attempt+'/native/stage-'+stage+'/16-before-retry.json');
 const after=await read(attempt+'/native/stage-'+stage+'/21-after-retry.json');
 const retry=await read(attempt+'/native/stage-'+stage+'/19-retry-selection.json');
 const complete=await read(attempt+'/native/stage-'+stage+'/14-complete.json');
 const receipts={};
 for(const [name,filename,expected,original] of [['source',f.sourceRolloverPath,f.sourceRolloverAfter,f.sourceRolloverBefore],['destination',f.destinationRolloverPath,f.destinationRollover,f.destinationRolloverBefore]]){
  const actual=await read(filename);
  receipts[name]={exactExpected:eq(actual,expected),provenancePreserved:['schemaVersion','date','planId','operationId','completedAt'].every(k=>eq(actual[k],original[k])),removedIds:actual.removedIds};
 }
 const retryOutput=JSON.parse(retry.stdout);
 const check={stage,observation,receipts,retryExit:retry.exitCode,retryOutput,authoritativeRetryUnchanged:eq(before.files,after.files),completionExit:complete.exitCode,completion:JSON.parse(complete.stdout)};
 check.pass=observation.verified && observation.otherPreserved && observation.prosePreserved && observation.rolloverPreserved && observation.checkpointRemoved && Object.values(receipts).every(r=>r.exactExpected&&r.provenancePreserved) && check.authoritativeRetryUnchanged && retry.exitCode===0 && retryOutput.effects.length===0 && retryOutput.conflicts.length===0 && complete.exitCode===0 && check.completion.status==='completed';
 checks.push(check);
 await save(attempt+'/stage-'+stage+'-after-inspection.json',check);
 await cp(vaultRoot,attempt+'/stage-'+stage+'-after',{recursive:true,errorOnExist:true,force:false});
}
const current=await inventory();
const old=await read(attempt+'/historical-evidence-hashes.json');
const changedHistory=[];
for(const [p,hash] of Object.entries(old)){try{if(sha(await readFile(p))!==hash)changedHistory.push(p)}catch{changedHistory.push(p)}}
const changedCandidate=Object.keys({...freeze.candidate.files,...current.files}).filter(p=>freeze.candidate.files[p]!==current.files[p]);
const reviewPrefix=(await readFile('docs/2026.10.13-plugin-review.md')).subarray(0,(await readFile(attempt+'/2026.10.13-plugin-review.md.before')).length).equals(await readFile(attempt+'/2026.10.13-plugin-review.md.before'));
const result={observedAt:new Date().toISOString(),candidate:freeze.head,head:execFileSync('git',['rev-parse','HEAD'],{encoding:'utf8'}).trim(),diffFromCandidate:execFileSync('git',['diff','--stat',freeze.head,'HEAD'],{encoding:'utf8'}),inventoryHash:current.hash,files:Object.keys(current.files).length,changedCandidate,historicalFiles:Object.keys(old).length,changedHistory,reviewPrefixPreserved:reviewPrefix,stages:checks.map(x=>({stage:x.stage,pass:x.pass})),pass:checks.every(x=>x.pass)&&!changedCandidate.length&&!changedHistory.length&&reviewPrefix};
await save(attempt+'/independent-verification.json',result);
console.log(JSON.stringify(result,null,2));
if(!result.pass)process.exitCode=1;
